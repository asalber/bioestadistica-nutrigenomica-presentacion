% Hoja
\newcommand{\hoja}[4]{% x, y, ángulo, escala
  \begin{scope}[shift={(#1,#2)}, rotate=#3, scale=#4]
    \fill[hojaverde!85] (0,0) .. controls (0.5,0.45) and (1.3,0.45) .. (1.8,0)
                        .. controls (1.3,-0.45) and (0.5,-0.45) .. (0,0);
    \draw[white, line width=0.6pt] (0.05,0) -- (1.6,0);
  \end{scope}}
% Fruta (círculo con brillo y rabito)
\newcommand{\fruta}[4]{% x, y, radio, color
  \fill[#4] (#1,#2) circle (#3);
  \fill[white, opacity=0.35] ($(#1,#2)+(-0.35*#3,0.35*#3)$) circle (0.28*#3);
  \draw[ceudark, line width=1pt] ($(#1,#2)+(0,#3)$) -- ++(0.1,0.3*#3);}
